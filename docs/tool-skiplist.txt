# tool-skiplist — backlog rows /create-next-tool should NOT build.
#
# Format:  <slug>  # reason
# Blank lines and lines starting with # are ignored. scripts/pick-next-tool.py
# reads this; the build agent appends a line when it spots a semantic near-dup
# mid-run (the exact-slug picker can't catch those — see SKILL.md "Known
# limitation"). Keep reasons short and concrete.
#
# Only list CONFIRMED duplicates of an existing blocks/<slug>/ tool, or rows that
# genuinely cannot be built in gizza's browser-local pure-Rust+ffmpeg model and
# aren't better handled by the picker's out-of-model defer. Do NOT list tools
# that merely share a word with an existing one (age-calculator is NOT a dup of
# calculator; http-request-builder constructs bytes and does NOT send, unlike
# http-request) — those are real, distinct tools.

pdf-to-text            # duplicate of blocks/pdf-extract-text (same: extract text from a PDF)
convert-image-format   # duplicate of blocks/image-convert (same: re-encode image to another format)
image-vectorize        # duplicate of blocks/vectorize (same: trace a raster image into SVG)
image-to-webp          # duplicate of blocks/image-convert (already does format=webp with quality)
jpg-compressor         # duplicate of blocks/image-compress (re-encodes jpg at a quality, keeps dimensions)

# Architecture-fit skips: need MORE THAN ONE media input. The page driver
# (site/tool.js) handles a single source="file" upload, and the descriptor
# Input enum + resolve_source/dispatch_ffmpeg model one media input — so a tool
# that combines two uploads doesn't fit without a multi-input architecture change.
add-audio-to-video     # needs two media inputs (video + audio); gizza ffmpeg model is single-input

# Two-opposite-IO-shapes: a single descriptor has ONE Input and one result shape.
# A bidirectional file<->text converter is really two tools (encode: Input::File
# -> text; decode: text param -> file envelope) — Input::File forces a required
# url/ref oneOf, which the decode direction has no use for.
base64-file-converter  # bidirectional file<->base64 = two IO shapes; split into base64-encode-file / base64-decode-file
